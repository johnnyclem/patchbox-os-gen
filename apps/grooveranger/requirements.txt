# GrooveRanger runtime dependencies (floors, not pins — see
# requirements.lock for what the image actually installs).

# The sequencer core is stdlib only (plus the vendored rangerkit): patterns,
# schedules, conditions, the song chain and the engine all run and test on a
# box with no audio, no MIDI and no display. numpy is the sampler and the FX
# bus; sounddevice is only the device layer, and its absence degrades to the
# silent null backend (which is exactly what CI runs).

mido>=1.3            # MIDI message model
python-rtmidi>=1.5   # mido's ALSA sequencer backend
alsa-midi>=1.0          ; platform_system == "Linux"  # native ALSA seq
numpy>=1.24          # sampler voices + FX bus blocks
sounddevice>=0.4     # PortAudio out for the internal endpoint
